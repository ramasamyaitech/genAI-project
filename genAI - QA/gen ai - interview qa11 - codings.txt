How do you limit the content size during document chunking?


def split_into_chunks(text, max_chars):
    paragraphs = text.split("\n\n")

    chunks = []
    current_chunk = ""

    for para in paragraphs:
        para = para.strip()

        if not para:
            continue

        # If paragraph itself is bigger than max_chars
        if len(para) > max_chars:

            # Save existing chunk first
            if current_chunk:
                chunks.append(current_chunk.strip())
                current_chunk = ""

            # Split large paragraph
            for i in range(0, len(para), max_chars):
                chunks.append(para[i:i + max_chars])

        else:
            # Check if paragraph can be added to current chunk
            if len(current_chunk) + len(para) + (2 if current_chunk else 0) <= max_chars:

                if current_chunk:
                    current_chunk += "\n\n" + para
                else:
                    current_chunk = para

            else:
                # Store previous chunk
                if current_chunk:
                    chunks.append(current_chunk)

                current_chunk = para

    # Add remaining chunk
    if current_chunk:
        chunks.append(current_chunk)

    return chunks


text = """Lorem ipsum dolor sit amet, consectetur adipiscing elit, sed do eiusmod tempor incididunt ut labore et dolore magna aliqua. chunks = split_into_chunks(text, 50)
